function closePosition(id) { const t = state.trades.find(x=>x.trade_id===id); if (!t) return showToast('未找到该持仓','error'); const refPrice = state.quotes?.[t.stock_code]?.price; showModal(`
清仓确认
${t.stock_name}（${t.stock_code}）
${t.position} 股
${(t.avg_cost||t.buy_price||0).toFixed(2)} 元
${refPrice?`
${refPrice.toFixed(2)} 元
`:''}
`); } async function doClosePosition(id) { const t = state.trades.find(x=>x.trade_id===id); if (!t) return showToast('未找到该持仓','error'); const price = parseFloat($('#cp_sellPrice').value); const reason = ($('#cp_reason')||{}).value||''; if (!price || price<=0) return showToast('请输入有效的卖出价','error'); closeModal(); try { const r = await api(`/api/trades/${id}/reduce-position`,{method:'POST',body:JSON.stringify({price,qty:t.position,reason:'清仓'})}); if (!r.success) { showToast('清仓失败: '+r.error,'error'); return; } showToast('清仓成功','success'); state.riskCheck = null; await loadAll(); renderTab('trades'); } catch(e){ showToast('清仓失败','error'); } }
